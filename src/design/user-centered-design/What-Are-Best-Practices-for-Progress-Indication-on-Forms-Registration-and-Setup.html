التصميم المتمحور حول المستخدم User-Centered Design

ما هي أفضل الممارسات للإشارة إلى التقدم Progress Indication في النماذج Forms والتسجيل Registration والإعداد Setup ؟

نظري

ملاحظة: قد تستخدم بعض الأمثلة التفاعلية أكواد CSS و JavaScript لم تتعلمها بعد. لا تقلق بشأن محاولة فهم الكود بالكامل؛ فالهدف من هذه الأمثلة هو عرض نماذج لمفاهيم التصميم هذه، لتتمكن من فهم آلية عملها بشكل أفضل.

الإشارة إلى التقدم هي طريقة لعرض مدى تقدم المستخدم في عملية ما. يمكن استخدامها في النماذج، والتسجيل، وعمليات الإعداد. الهدف هو مساعدة المستخدم على فهم موقعه في العملية وكم تبقى عليه من خطوات.

على سبيل المثال، يمكنك استخدام شريط إشارة التقدم لتوضيح ما تبقى للمستخدم عند ملء النماذج. لا تريد أن تضع المستخدم في موقف يحتاج فيه لملء نموذج طويل ولا يعرف كم خطوة متبقية. الشفافية مهمة ليعرف المستخدم إذا كان لديه وقت كافٍ للجلوس وإكمال النموذج أو إذا كان عليه العودة لاحقًا.

index.js


const form = document.getElementById('multiStepForm');
const steps = form.querySelectorAll('.form-step');
const progressBar = form.querySelector('.progress-bar');
const progressText = form.querySelector('.progress-text');
const totalSteps = steps.length;

let currentStep = 0;

function updateProgress() {
  const percent = ((currentStep + 1) / totalSteps) * 100;
  progressBar.style.width = percent + '%';
  progressText.textContent = `الخطوة ${currentStep + 1} من ${totalSteps}`;
}

function showStep(index) {
  steps.forEach((step, i) => {
    step.classList.toggle('active', i === index);
  });
  updateProgress();
}

form.querySelectorAll('.next-btn').forEach(btn => btn.addEventListener('click', () => {
  if (currentStep < totalSteps - 1) {
    currentStep++;
    showStep(currentStep);
  }
}));

form.querySelectorAll('.prev-btn').forEach(btn => btn.addEventListener('click', () => {
  if (currentStep > 0) {
    currentStep--;
    showStep(currentStep);
  }
}));

showStep(currentStep);

form.addEventListener('submit', e => {
  e.preventDefault();
  alert('تم إرسال النموذج!');
});

معاينة:

عند تصميم قسم إشارة التقدم، هناك بعض أفضل الممارسات التي يجب مراعاتها. الاعتبار الأول هو الحفاظ على البساطة. لا تريد إرباك المستخدم بمعلومات كثيرة تجعله يشعر بالإحباط ويغادر الموقع.

الاعتبار الثاني هو جعل العودة إلى الخطوات السابقة ممكنة. هذا مهم لأن المستخدمين قد يرغبون في مراجعة إجاباتهم السابقة أو تعديلها.

اعتبار آخر هو جعل قسم إشارة التقدم سهل العثور عليه. إذا لم يجد المستخدم القسم، فلن يعرف مدى تقدمه في العملية.

الاعتبار الأخير هو وجود عناوين أقسام واضحة، أو نسب مئوية، أو خطوات. إذا كان لديك فقط شريط تقدم بدون سياق، فلن يعرف المستخدم ما يعنيه.

index.js


const form = document.getElementById('progressForm');
const steps = form.querySelectorAll('.form-step');
const progressBar = form.querySelector('.progress-bar');
const currentStepSpan = document.getElementById('currentStep');
const totalStepsSpan = document.getElementById('totalSteps');
const percentageSpan = document.getElementById('percentage');

const totalSteps = steps.length;
let currentStep = 0;

totalStepsSpan.textContent = totalSteps;

function updateProgress() {
  const percent = Math.round(((currentStep + 1) / totalSteps) * 100);
  progressBar.style.width = percent + '%';
  currentStepSpan.textContent = currentStep + 1;
  percentageSpan.textContent = percent + '%';

  // Update ARIA attribute
  form.querySelector('.progress-container').setAttribute('aria-valuenow', percent);
}

function showStep(index) {
  steps.forEach((step, i) => {
    step.classList.toggle('active', i === index);
  });
  updateProgress();
}

form.querySelectorAll('.next-btn').forEach(btn => btn.addEventListener('click', () => {
  if (currentStep < totalSteps - 1) {
    currentStep++;
    showStep(currentStep);
  }
}));

form.querySelectorAll('.prev-btn').forEach(btn => btn.addEventListener('click', () => {
  if (currentStep > 0) {
    currentStep--;
    showStep(currentStep);
  }
}));

form.addEventListener('submit', e => {
  e.preventDefault();
  alert('تم إرسال النموذج! شكرًا لك!');
});

showStep(currentStep);

معاينة:

هذه بعض أفضل الممارسات التي يجب مراعاتها عند تصميم إشارة التقدم في النماذج، والتسجيل، وعمليات الإعداد. ادرس بعض الأمثلة في المواقع الكبيرة لترى كيف ينفذون إشارة التقدم. ثم اختبر تصميمك مع مستخدمين حقيقيين لترى مدى فعاليته.

أسئلة :

ما هو الهدف الأساسي من استخدام إشارة التقدم في النماذج أو عمليات التسجيل؟

1. لترفيه المستخدمين أثناء إكمال النموذج. ❌

2. لإعلام المستخدمين بمدى تقدمهم في العملية وما تبقى عليهم فعله. ✅

3. لتقليل عدد الخطوات في العملية. ❌

4. لتسريع عملية إرسال النموذج. ❌


لماذا من المهم السماح للمستخدمين بالعودة إلى الخطوات السابقة في نموذج أو عملية تسجيل؟

1. للسماح للمستخدمين بتغيير إجاباتهم أو مراجعة المدخلات السابقة. ✅

2. لجعل النموذج أكثر تعقيدًا. ❌

3. لتقليل عدد الخطوات في العملية. ❌

4. لزيادة الوقت الذي يقضيه المستخدمون في الموقع. ❌


ما الذي يجب مراعاته عند وضع قسم إشارة التقدم في صفحة ويب؟

1. يجب أن يكون صغيرًا قدر الإمكان لتجنب تشتيت المستخدم. ❌

2. يجب أن يكون سهل العثور عليه وموسومًا بوضوح مع سياق مثل عناوين الأقسام أو النسب المئوية. ✅

3. يجب أن يكون مخفيًا حتى يصل المستخدم إلى الخطوة النهائية. ❌

4. يجب أن يتضمن فقط شريط تقدم بصري بدون نص. ❌
